# Guest functions the runtime reimplements instead of recompiling (one symbol name per line).
# Each one needs an HLE(name) definition in recomp/runtime/hle.c.

# Console output: OSReport -> vprintf -> stdout -> __write_console. On retail hardware this writes
# to the dev-kit UART, so the output is lost; print it on the host instead.
__write_console
__TRK_write_console

# Audio DSP bring-up: uploads a boot task and polls DSP mailboxes. Audio will be a host backend.
__OSInitAudioSystem

# Configures BATs and the memory interface via real mode + rfi. Host memory needs no protection setup.
__OSInitMemoryProtection

# OSInit turns on the FPU's non-IEEE mode (FPSCR NI: denormal results flush to zero). The host FPU must
# do the same (flush-to-zero and denormals-are-zero), or game code that relies on tiny values becoming
# exactly 0 goes wrong: see rt_fp_apply_mode in hle.c.
PPCSetFpNonIEEEMode


# Thread switching: each guest thread runs on its own host thread (see threads.c).
OSSaveContext
OSLoadContext

# No GPU yet, so drawing is always already done. (Waiting would need the PE finish interrupt.)
GXDrawDone

# Controllers come from the host (SDL later); for now port 1 is a connected pad with nothing pressed.
PADRead

# REL modules: record where the SDK linked each module so its recompiled code can find it (runtime.c).
OSNotifyLink
OSNotifyUnlink

# Memory card: slot A is a 59-block card stored as .gci files in user/GF7E01/card_a (card.c).
CARDProbeEx
CARDMountAsync
CARDUnmount
CARDGetResultCode
CARDCheckAsync
CARDCheckExAsync
CARDGetSerialNo
CARDFreeBlocks
CARDOpen
CARDClose
CARDCreateAsync
CARDDeleteAsync
CARDFormatAsync
CARDRead
CARDReadAsync
CARDWrite
CARDWriteAsync
CARDGetStatus
CARDSetStatus
CARDSetStatusAsync

# Boot ROM reads: the SDK loads the system font from the IPL ROM (OSInitFont/OSLoadFont).
__OSReadROM

# Entry hooks: the recompiled function runs, but calls rt_hook_XXXXXXXX(c) first.
# fn_8005F470(state, stage, ...) starts a stage (an index into the stage table at 0x80223560): logged.
entry fn_8005F470
# OSExitThread marks the current thread as exiting, so its host thread ends once it switches away.
entry OSExitThread
# fn_8001D5B4(stage) sets the current stage: copies its stage-table record into the game info at
# 0x802A2B60 (+0x13C stage, +0x13F play type, which picks the gameplay module). --stage swaps it here.
entry fn_8001D5B4
# fn_8000ECC8(ctl): the game's main state machine; ctl+0x34 is its state (37 states, jump table at
# 0x8021D0DC). State 21 ends a stage part and picks the next one from the table at 0x8021CF38.
# Testing aids in hle.c: SFA_GAME_STATES=1 logs state changes; --skip-part jumps to the next part.
entry fn_8000ECC8
# PSVECNormalize(src, dst): testing aid in hle.c (SFA_VEC_DEBUG=1 logs inputs too small to normalise).
entry PSVECNormalize
# fn_8003AEB0(): called once a frame by the main loop; derives the frame period (0x80385B04, VI units of
# 0x444) and the time-step constants from the target frame rate at 0x80385AF4 (60 or 30). The hook in
# hle.c logs rate changes (SFA_FPS_LOG=1) and can force 60 (SFA_60FPS=1).
entry fn_8003AEB0
# VIWaitForRetrace: the game's frame end always waits for the next retrace after swapping buffers;
# after a frame that ran late, hle.c skips that wait (the SDK's loop otherwise), so a frame 1 ms late
# isn't held for a whole extra 16.7 ms.
VIWaitForRetrace
# fn_800FA400: cubic Bezier curve length, summed over 1,000 steps. hle.c does the same float operations natively;
# when Mission 1's fleet is set up the game measures ~680 curves at once, which made a 25-42 ms frame.
fn_800FA400
